Two Sum

Given an array of integers and a target, return indices of the two numbers that add up to the target.

# Python 3

def twoSum(nums, target):
    n = {}
    for i, num in enumerate(nums):
        if target - num in n:
            return [n[target-num], i]
        n[num] = i
